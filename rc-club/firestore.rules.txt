rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /rcClub/data {
      // Students and other visitors can see the championship.
      allow read: if true;

      // Replace the email below with the same teacher Google account used in firebase-config.js.
      // Only that authenticated account can change students, races, results or points.
      allow write: if request.auth != null
                   && request.auth.token.email == "PASTE_TEACHER_GOOGLE_EMAIL";
    }
  }
}
